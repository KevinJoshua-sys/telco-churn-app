import streamlit as st
import pandas as pd
import joblib

# Load trained model and features list
model = joblib.load('model.pkl')
model_columns = joblib.load('model_columns.pkl')

st.title("Customer Churn Prediction Dashboard")
st.write("Adjust customer parameters to predict real-time churn risk.")

# User Inputs
tenure = st.slider("Tenure (Months)", min_value=0, max_value=72, value=12)
monthly_charges = st.number_input("Monthly Charges ($)", min_value=18.0, max_value=120.0, value=65.0)
total_charges = st.number_input("Total Charges ($)", min_value=0.0, max_value=9000.0, value=780.0)

contract = st.selectbox("Contract Type", ["Month-to-month", "One year", "Two year"])
internet_service = st.selectbox("Internet Service", ["DSL", "Fiber optic", "No"])
payment_method = st.selectbox("Payment Method", [
    "Electronic check", "Mailed check", "Bank transfer (automatic)", "Credit card (automatic)"
])

if st.button("Calculate Churn Probability"):
    # Build a raw input dataframe
    raw_data = {
        'tenure': tenure,
        'MonthlyCharges': monthly_charges,
        'TotalCharges': total_charges,
        'Contract': contract,
        'InternetService': internet_service,
        'PaymentMethod': payment_method
    }
    
    df_input = pd.DataFrame([raw_data])
    
    # One-hot encode inputs
    df_encoded = pd.get_dummies(df_input)
    
    # Align encoded input columns with training columns
    df_final = pd.DataFrame(columns=model_columns)
    for col in model_columns:
        if col in df_encoded.columns:
            df_final[col] = df_encoded[col]
        else:
            df_final[col] = 0
            
    # Predict
    prediction = model.predict(df_final)[0]
    prob = model.predict_proba(df_final)[0][1] * 100

    if prediction == 1:
        st.error(f"High Risk of Churn! Probability: {prob:.1f}%")
    else:
        st.success(f"Low Risk of Churn! Probability: {prob:.1f}%")
